cmake_minimum_required(VERSION 3.27)
project(OpenRVPreferenceLifetimeTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Test)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(OPENRV_LIFETIME_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../.." CACHE PATH "Actual source after applying the OpenRV patches")
option(OPENRV_LIFETIME_LEGACY_NEGATIVE "Compile the pre-fix raw-cache negative control; its test must fail" OFF)
set(_arguments --source "${OPENRV_LIFETIME_SOURCE}" --output "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(OPENRV_LIFETIME_LEGACY_NEGATIVE)
  list(APPEND _arguments --legacy-negative)
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/extract_methods.py" ${_arguments}
  RESULT_VARIABLE _result OUTPUT_VARIABLE _out ERROR_VARIABLE _error)
if(NOT _result EQUAL 0)
  message(FATAL_ERROR "Cannot extract the actual preference lifetime methods: ${_error}")
endif()
add_executable(pref_dialog_lifetime_tests test_pref_dialog_lifetime.cpp)
target_include_directories(pref_dialog_lifetime_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(pref_dialog_lifetime_tests PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Test)
if(OPENRV_LIFETIME_LEGACY_NEGATIVE)
  target_compile_definitions(pref_dialog_lifetime_tests PRIVATE OPENRV_LIFETIME_LEGACY_NEGATIVE)
endif()
enable_testing()
add_test(NAME pref_dialog_lifetime COMMAND pref_dialog_lifetime_tests)
set_tests_properties(pref_dialog_lifetime PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen"
  ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
