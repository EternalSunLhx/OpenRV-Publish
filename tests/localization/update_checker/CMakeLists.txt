cmake_minimum_required(VERSION 3.27)
project(OpenRVUpdateCheckerTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core Gui Widgets Network Test)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Gui Widgets Network Test)
set(_source "${CMAKE_CURRENT_LIST_DIR}/../../../src/lib/app/RvCommon")
if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/../../../overlay/src/lib/app/RvCommon/RvUpdateChecker.cpp")
  set(_source "${CMAKE_CURRENT_LIST_DIR}/../../../overlay/src/lib/app/RvCommon")
endif()
set(RV_MAJOR_VERSION 4)
set(RV_MINOR_VERSION 1)
set(RV_REVISION_NUMBER 0)
set(RV_TEST_RELEASE_TAG "" CACHE STRING "Optional official or legacy released-build fixture")
set(RV_TEST_BUILD_NUMBER "0" CACHE STRING "Internal official build number; zero for development or legacy")
set(RV_TEST_NATIVE_VERSION "" CACHE STRING "Optional native CMake version distinct from the official release version")
if(NOT RV_TEST_BUILD_NUMBER MATCHES "^(0|[1-9][0-9]*)$")
  message(FATAL_ERROR "Invalid fixture build number")
endif()
set(_rv_localized_tag "${RV_TEST_RELEASE_TAG}")
set(_rv_localized_build_number "${RV_TEST_BUILD_NUMBER}")
if(_rv_localized_tag STREQUAL "")
  if(NOT RV_TEST_BUILD_NUMBER STREQUAL "0")
    message(FATAL_ERROR "Development fixture must not claim an official build number")
  endif()
  set(_rv_localized_version "4.1.0-dev.0123abcd")
elseif(_rv_localized_tag MATCHES "^v([0-9]+)\\.([0-9]+)\\.([0-9]+)-zh\\.([1-9][0-9]*)$")
  if(NOT RV_TEST_BUILD_NUMBER STREQUAL "0")
    message(FATAL_ERROR "Legacy fixture must not claim an official build number")
  endif()
  set(RV_MAJOR_VERSION "${CMAKE_MATCH_1}")
  set(RV_MINOR_VERSION "${CMAKE_MATCH_2}")
  set(RV_REVISION_NUMBER "${CMAKE_MATCH_3}")
  string(SUBSTRING "${_rv_localized_tag}" 1 -1 _rv_localized_version)
elseif(_rv_localized_tag MATCHES "^v([0-9]+)\\.([0-9]+)\\.([0-9]+)$")
  set(RV_MAJOR_VERSION "${CMAKE_MATCH_1}")
  set(RV_MINOR_VERSION "${CMAKE_MATCH_2}")
  set(RV_REVISION_NUMBER "${CMAKE_MATCH_3}")
  if(RV_TEST_BUILD_NUMBER STREQUAL "0")
    message(FATAL_ERROR "Official fixture requires a positive build number")
  endif()
  string(SUBSTRING "${_rv_localized_tag}" 1 -1 _rv_localized_base_version)
  set(_rv_localized_version "${_rv_localized_base_version}+build.${RV_TEST_BUILD_NUMBER}")
else()
  message(FATAL_ERROR "Invalid fixture release tag")
endif()
set(CORE_VERSION "${RV_MAJOR_VERSION}.${RV_MINOR_VERSION}.${RV_REVISION_NUMBER}")
if(NOT RV_TEST_NATIVE_VERSION STREQUAL "")
  set(CORE_VERSION "${RV_TEST_NATIVE_VERSION}")
endif()
set(_rv_localized_major "${RV_MAJOR_VERSION}")
set(_rv_localized_minor "${RV_MINOR_VERSION}")
set(_rv_localized_patch "${RV_REVISION_NUMBER}")
configure_file("${_source}/RvCommon/RvLocalizedBuild.h.in" "generated/RvLocalizedBuild.h" @ONLY)
add_executable(update_checker_tests test_update_checker.cpp "${_source}/RvUpdateChecker.cpp" "${_source}/RvCommon/RvUpdateChecker.h")
target_include_directories(update_checker_tests PRIVATE "${_source}" "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(update_checker_tests PRIVATE Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::Gui
  Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::Network Qt${QT_VERSION_MAJOR}::Test)
if(MSVC)
  target_compile_options(update_checker_tests PRIVATE /utf-8)
endif()
enable_testing()
add_test(NAME update_checker COMMAND update_checker_tests)
set_tests_properties(update_checker PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
set_tests_properties(update_checker PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt${QT_VERSION_MAJOR}::Core>")
