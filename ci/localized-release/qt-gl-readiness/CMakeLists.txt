cmake_minimum_required(VERSION 3.24)
project(OpenRvHostedGlReadiness LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.5.3 EXACT REQUIRED COMPONENTS Core Gui)
get_filename_component(OPENRV_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(openrv-gl-readiness main.cpp "${OPENRV_SOURCE_ROOT}/src/pub/glew/glew.c")
target_include_directories(openrv-gl-readiness PRIVATE "${OPENRV_SOURCE_ROOT}/src/pub/glew")
target_compile_definitions(openrv-gl-readiness PRIVATE GLEW_STATIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(openrv-gl-readiness PRIVATE Qt6::Core Qt6::Gui opengl32)
