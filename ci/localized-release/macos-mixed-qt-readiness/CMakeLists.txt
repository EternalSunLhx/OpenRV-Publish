cmake_minimum_required(VERSION 3.24)
project(OpenRvMixedQtFixture LANGUAGES CXX)
if(NOT APPLE)
  message(FATAL_ERROR "This fixture requires native macOS Cocoa and dyld")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5.3 EXACT REQUIRED COMPONENTS Core Gui Widgets WebEngineCore WebEngineWidgets)
find_package(Python3 3.11 EXACT REQUIRED COMPONENTS Interpreter Development.Embed)
add_executable(openrv-mixed-qt MACOSX_BUNDLE main.cpp)
target_link_libraries(openrv-mixed-qt PRIVATE Qt6::Widgets Qt6::WebEngineCore Qt6::WebEngineWidgets Python3::Python)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/python-library.txt" CONTENT "$<TARGET_FILE:Python3::Python>\n")
set_target_properties(openrv-mixed-qt PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER org.openrv.MixedQtFixture)
